23/// Builds the helper type that prints a map key without the quotes a `str` would be formatted with.
24///
25/// A nameless struct or variant is formatted as a map, and its keys are the field names; the type is declared once per generated `fmt` body, so an enum with several such variants does not repeat it.
26#[inline]
27pub(crate) fn create_raw_string_type(ident: &Ident, formatter: &Ident) -> proc_macro2::TokenStream {
28    quote_mixed!(
29        #[allow(non_camel_case_types)]
30        struct #ident(&'static ::core::primitive::str);
31
32        impl ::core::fmt::Debug for #ident {
33            #[inline]
34            fn fmt(&self, #formatter: &mut ::core::fmt::Formatter<'_>) -> ::core::fmt::Result {
35                #formatter.write_str(self.0)
36            }
37        }
38    )
39}

41/// Builds the statement that starts a map builder; the caller has to emit [`create_raw_string_type`] once in the same block.
42#[inline]
43pub(crate) fn create_debug_map_builder(
44    formatter: &Ident,
45    builder: &Ident,
46) -> proc_macro2::TokenStream {
47    quote_mixed!(let mut #builder = #formatter.debug_map();)
48}

50/// Wraps a field with a closure that keeps the source impl's bounds and `Self` scope.
51#[inline]
52pub(crate) fn create_format_arg(
53    bindings: &mut ValueBindings,
54    formatter: &Ident,
55    ident: &Ident,
56    field_ty: &Type,
57    format_method: &ExprPath,
58    field_expr: proc_macro2::TokenStream,
59) -> (Ident, proc_macro2::TokenStream) {
60    let value = bindings.binding("value");
61    let arg = bindings.binding("arg");
62    let tokens = quote_mixed!(
63        let #arg = {
64            #[allow(non_camel_case_types)]
65            struct #ident<'a, V: ?::core::marker::Sized, F>(&'a V, F);
66
67            impl<V: ?::core::marker::Sized, F> ::core::fmt::Debug for #ident<'_, V, F>
68            where
69                F: ::core::ops::Fn(&V, &mut ::core::fmt::Formatter<'_>) -> ::core::fmt::Result,
70            {
71                #[inline]
72                fn fmt(&self, #formatter: &mut ::core::fmt::Formatter<'_>) -> ::core::fmt::Result {
73                    (self.1)(self.0, #formatter)
74                }
75            }
76
77            #ident(#field_expr, |#value: &#field_ty, #formatter: &mut ::core::fmt::Formatter<'_>| #format_method(#value, #formatter))
78        };
79    );
80    (arg, tokens)
81}

83/// Keeps custom formatting methods visible to dead-code analysis under the final impl bounds.
84pub(crate) fn create_mark_method_used(
85    ast: &DeriveInput,
86    ctx: &TraitHandlerContext,
87    generics: &syn::Generics,
88    field_ty: &Type,
89    method: &ExprPath,
90) -> proc_macro2::TokenStream {
91    let marker = MethodMarker::new(ast, generics, field_ty, method);
92    let (impl_generics, _, where_clause) = marker.generics.split_for_impl();
93    let field_ty = &marker.field_ty;
94    let method = &marker.method;
95    let mut bindings = ctx.value_bindings(ast);
96    let marker_name = bindings.fresh("__educe_debug_method_used");
97    let value = bindings.binding("value");
98    let formatter = bindings.binding("formatter");
99
100    marker.wrap(
101        quote_mixed!(
102            fn #marker_name #impl_generics (
103                #value: &#field_ty,
104                #formatter: &mut ::core::fmt::Formatter<'_>,
105            ) -> ::core::fmt::Result #where_clause {
106                #method(#value, #formatter)
107            }
108        ),
109        &bindings,
110    )
111}
