15/// Uses a whole-value copy only when it does not add requirements to `Clone`.
16fn can_use_bitwise_copy(
17    ast: &DeriveInput,
18    ctx: &TraitHandlerContext,
19    traits: &[Trait],
20    has_custom_method: bool,
21) -> syn::Result<bool> {
22    #[cfg(feature = "Copy")]
23    {
24        if !traits.contains(&Trait::Copy) || has_custom_method {
25            return Ok(false);
26        }
27
28        // A custom `Copy` bound can limit lifetimes or unused const parameters even when the fields do not mention a type parameter.
29        if !ast.generics.params.is_empty() && super::copy::has_custom_bound(ctx)? {
30            return Ok(false);
31        }
32
33        let uses_generics = |field: &syn::Field| {
34            crate::common::r#type::type_uses_generic_params(&field.ty, &ast.generics.params)
35        };
36
37        Ok(match &ast.data {
38            Data::Struct(data) => !data.fields.iter().any(uses_generics),
39            Data::Enum(data) => {
40                !data.variants.iter().any(|variant| variant.fields.iter().any(uses_generics))
41            },
42            Data::Union(_) => false,
43        })
44    }
45    #[cfg(not(feature = "Copy"))]
46    {
47        let _ = (ast, ctx, traits, has_custom_method);
48        Ok(false)
49    }
50}

52/// Builds a module-level marker that references a field's custom clone method so it stays counted as used.
53pub(crate) fn create_mark_method_used(
54    ast: &DeriveInput,
55    ctx: &TraitHandlerContext,
56    generics: &Generics,
57    field_ty: &Type,
58    method: &ExprPath,
59) -> proc_macro2::TokenStream {
60    let marker = MethodMarker::new(ast, generics, field_ty, method);
61    let (impl_generics, _ty_generics, where_clause) = marker.generics.split_for_impl();
62    let field_ty = &marker.field_ty;
63    let method = &marker.method;
64    let mut bindings = ctx.value_bindings(ast);
65    let marker_name = bindings.fresh("__educe_clone_method_used");
66    let value = bindings.binding("value");
67
68    marker.wrap(
69        quote_mixed!(
70            fn #marker_name #impl_generics (
71                #value: &#field_ty,
72            ) -> #field_ty #where_clause {
73                #method(#value)
74            }
75        ),
76        &bindings,
77    )
78}

80/// Dispatches the `Clone` derive to the specialized handler for the shape of the input (struct, enum, or union).
81pub(crate) struct CloneHandler;
82
83impl TraitHandler for CloneHandler {
84    #[inline]
85    fn trait_meta_handler<'a>(
86        ast: &'a DeriveInput,
87        ctx: &mut TraitHandlerContext<'a>,
88        token_stream: &mut proc_macro2::TokenStream,
89        traits: &[Trait],
90        meta: &Meta,
91    ) -> syn::Result<()> {
92        match ast.data {
93            Data::Struct(_) => clone_struct::CloneStructHandler::trait_meta_handler(
94                ast,
95                ctx,
96                token_stream,
97                traits,
98                meta,
99            ),
100            Data::Enum(_) => clone_enum::CloneEnumHandler::trait_meta_handler(
101                ast,
102                ctx,
103                token_stream,
104                traits,
105                meta,
106            ),
107            Data::Union(_) => clone_union::CloneUnionHandler::trait_meta_handler(
108                ast,
109                ctx,
110                token_stream,
111                traits,
112                meta,
113            ),
114        }
115    }
