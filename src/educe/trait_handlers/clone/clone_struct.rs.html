17/// Generates the `Clone` implementation for a struct.
18pub(crate) struct CloneStructHandler;
19
20impl TraitHandler for CloneStructHandler {
21    #[inline]
22    fn trait_meta_handler<'a>(
23        ast: &'a DeriveInput,
24        ctx: &mut TraitHandlerContext<'a>,
25        token_stream: &mut proc_macro2::TokenStream,
26        traits: &[Trait],
27        meta: &Meta,
28    ) -> syn::Result<()> {
29        let generated_impl_attributes =
30            crate::common::attributes::generated_impl_attributes(&ast.attrs);
31
32        let type_attribute = TypeAttributeBuilder {
33            enable_flag: true, enable_bound: true
34        }
35        .build_from_clone_meta(meta)?;
36
37        let mut bound: WherePredicates = Punctuated::new();
38
39        // Custom clone methods are referenced only inside the derived impl body, which dead-code analysis skips, so each one is collected here and later re-referenced by a marker item.
40        let mut mark_fields: Vec<(&Type, ExprPath)> = Vec::new();
41
42        // A `#[repr(packed)]` type reads every cloned field through a copy, so those field types additionally have to be `Copy`.
43        let is_packed = is_packed(&ast.attrs);
44        let mut copy_types: Vec<&Type> = Vec::new();
45        let this = quote_mixed!(self);
46        let mut bindings = ctx.value_bindings(ast);
47
48        let mut clone_token_stream = proc_macro2::TokenStream::new();
49        let mut clone_from_token_stream = proc_macro2::TokenStream::new();
50
51        if let Data::Struct(data) = &ast.data {
52            let mut fields: Vec<(&Field, FieldAttribute)> = Vec::new();
53
54            for field in data.fields.iter() {
55                let field_attribute = FieldAttributeBuilder {
56                    enable_method: true
57                }
58                .build_from_attributes(&field.attrs, traits)?;
59
60                fields.push((field, field_attribute));
61            }
62
63            let has_custom_method =
64                fields.iter().any(|(_, field_attribute)| field_attribute.method.is_some());
65
66            let use_bitwise_copy =
67                super::can_use_bitwise_copy(ast, ctx, traits, has_custom_method)?;
68
69            let mut clone_types: Vec<&Type> = Vec::new();
70
71            if use_bitwise_copy {
72                // A whole-value copy needs no field-wise body or field bounds.
73                clone_token_stream.extend(quote_mixed!(*self));
74            } else {
75                let source = bindings.binding("source");
76                let source_receiver = quote_mixed!(#source);
77                let mut fields_token_stream = proc_macro2::TokenStream::new();
78                let mut clone_from_body_token_stream = proc_macro2::TokenStream::new();
79
80                if fields.is_empty() {
81                    clone_from_body_token_stream.extend(quote_mixed!(let _ = #source;));
82                } else {
83                    for (index, (field, field_attribute)) in fields.into_iter().enumerate() {
84                        let field_name =
85                            IdentOrIndex::from_ident_with_index(field.ident.as_ref(), index);
86
87                        if is_packed {
88                            copy_types.push(&field.ty);
89                        }
90
91                        let self_ref = borrow_field(is_packed, &this, &field_name);
92                        let source_ref = borrow_field(is_packed, &source_receiver, &field_name);
93
94                        let value = if let Some(clone) = field_attribute.method.as_ref() {
95                            mark_fields.push((&field.ty, clone.clone()));
96
97                            clone_from_body_token_stream
98                                .extend(quote_mixed!(self.#field_name = #clone(#source_ref);));
99
100                            quote_mixed!(#clone(#self_ref))
101                        } else {
102                            clone_types.push(&field.ty);
103
104                            // A packed field cannot be borrowed mutably either, so the cloned value is assigned back instead of being cloned in place.
105                            clone_from_body_token_stream.extend(if is_packed {
106                                quote_mixed!(self.#field_name = ::core::clone::Clone::clone(#source_ref);)
107                            } else {
108                                quote_mixed!( ::core::clone::Clone::clone_from(&mut self.#field_name, #source_ref); )
109                            });
110
111                            quote_mixed!(::core::clone::Clone::clone(#self_ref))
112                        };
113
114                        fields_token_stream.extend(field_name.to_field(&value));
115                    }
116                }
117
118                clone_token_stream.extend(match &data.fields {
119                    Fields::Unit => quote_mixed!(Self),
120                    Fields::Named(_) => quote_mixed!(Self { #fields_token_stream }),
121                    Fields::Unnamed(_) => quote_mixed!(Self ( #fields_token_stream )),
122                });
123
124                clone_from_token_stream.extend(clone_from_body_token_stream);
125            }
126
127            let packed_copy_predicates = if is_packed {
128                type_attribute.bound.packed_copy_predicates(&ast.generics, &copy_types, &ast.ident)
129            } else {
130                WherePredicates::new()
131            };
132
133            // The bound trait is always `Clone`; the `Copy` impl is emitted by the `Copy` handler with its own bounds.
134            bound = type_attribute.bound.into_where_predicates_by_generic_parameters_check_types(
135                &ast.generics,
136                &syn::parse2(quote_mixed!(::core::clone::Clone)).unwrap(),
137                &clone_types,
138                &ast.ident,
139                &BOUND_EXCEPTIONS_CLONE.with_custom_fields(!mark_fields.is_empty()),
140            );
141
142            extend_where_predicates(&mut bound, packed_copy_predicates);
143
144            ctx.record(Trait::Clone, &bound);
145        }
146
147        let clone_from_fn_token_stream = if clone_from_token_stream.is_empty() {
148            None
149        } else {
150            let source = bindings.binding("source");
151            Some(quote_mixed! {
152                #[inline]
153                fn clone_from(&mut self, #source: &Self) {
154                    #clone_from_token_stream
155                }
156            })
157        };
158
159        let ident = &ast.ident;
160
161        let generics = crate::common::generics::with_predicates(ast.generics.clone(), bound);
162
163        let (impl_generics, ty_generics, where_clause) = generics.split_for_impl();
164
165        token_stream.extend(bindings.wrap(quote_mixed! {
166            #generated_impl_attributes
167            impl #impl_generics ::core::clone::Clone for #ident #ty_generics #where_clause {
168                #[inline]
169                fn clone(&self) -> Self {
170                    #clone_token_stream
171                }
172
173                #clone_from_fn_token_stream
174            }
175        }));
176
177        for (field_ty, method) in &mark_fields {
178            token_stream
179                .extend(super::create_mark_method_used(ast, ctx, &generics, field_ty, method));
180        }
181
182        Ok(())
183    }
184}
