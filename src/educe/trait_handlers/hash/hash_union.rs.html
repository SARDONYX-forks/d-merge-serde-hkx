10/// Generates the `Hash` implementation for a union.
11pub(crate) struct HashUnionHandler;
12
13impl TraitHandler for HashUnionHandler {
14    #[inline]
15    fn trait_meta_handler<'a>(
16        ast: &'a DeriveInput,
17        ctx: &mut TraitHandlerContext<'a>,
18        token_stream: &mut proc_macro2::TokenStream,
19        traits: &[Trait],
20        meta: &Meta,
21    ) -> syn::Result<()> {
22        let generated_impl_attributes =
23            crate::common::attributes::generated_impl_attributes(&ast.attrs);
24        let mut bindings = ctx.value_bindings(ast);
25        let state = bindings.binding("state");
26
27        let type_attribute =
28            TypeAttributeBuilder {
29                enable_flag: true, enable_unsafe: true, enable_bound: true
30            }
31            .build_from_hash_meta(meta)?;
32
33        if !type_attribute.has_unsafe {
34            return Err(crate::panic::union_without_unsafe(meta));
35        }
36
37        if let Data::Union(data) = &ast.data {
38            for field in data.fields.named.iter() {
39                let _ = FieldAttributeBuilder {
40                    enable_ignore: false, enable_method: false
41                }
42                .build_from_attributes(&field.attrs, traits)?;
43            }
44        }
45
46        let ident = &ast.ident;
47        let hasher_ident = bindings.fresh("H");
48
49        // Hashing the storage as bytes needs nothing from the field types, so the automatic bound stays empty and only an explicit one contributes predicates.
50        let bound = type_attribute.bound.into_where_predicates_by_generic_parameters_check_types(
51            &ast.generics,
52            &syn::parse2(quote_mixed!(::core::hash::Hash)).unwrap(),
53            &[],
54            &ast.ident,
55            &BOUND_EXCEPTIONS_HASH,
56        );
57
58        let generics = crate::common::generics::with_predicates(ast.generics.clone(), bound);
59
60        let (impl_generics, ty_generics, where_clause) = generics.split_for_impl();
61
62        let data = bindings.binding("data");
63        let bytes = union_bytes(&data, &quote_mixed!(self));
64
65        token_stream.extend(bindings.wrap(quote_mixed! {
66            #generated_impl_attributes
67            impl #impl_generics ::core::hash::Hash for #ident #ty_generics #where_clause {
68                #[inline]
69                fn hash<#hasher_ident: ::core::hash::Hasher>(&self, #state: &mut #hasher_ident) {
70                    #bytes
71
72                    ::core::hash::Hash::hash(#data, #state)
73                }
74            }
75        }));
76
77        Ok(())
78    }
79}
