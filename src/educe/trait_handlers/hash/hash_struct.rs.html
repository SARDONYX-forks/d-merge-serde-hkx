19/// Generates the `Hash` implementation for a struct.
20pub(crate) struct HashStructHandler;
21
22impl TraitHandler for HashStructHandler {
23    #[inline]
24    fn trait_meta_handler<'a>(
25        ast: &'a DeriveInput,
26        ctx: &mut TraitHandlerContext<'a>,
27        token_stream: &mut proc_macro2::TokenStream,
28        traits: &[Trait],
29        meta: &Meta,
30    ) -> syn::Result<()> {
31        let generated_impl_attributes =
32            crate::common::attributes::generated_impl_attributes(&ast.attrs);
33        let mut bindings = ctx.value_bindings(ast);
34        let state = bindings.binding("state");
35
36        let type_attribute =
37            TypeAttributeBuilder {
38                enable_flag: true, enable_unsafe: false, enable_bound: true
39            }
40            .build_from_hash_meta(meta)?;
41
42        let mut hash_types: Vec<&Type> = Vec::new();
43
44        // A `#[repr(packed)]` type reads every hashed field through a copy, so those field types additionally have to be `Copy`.
45        let is_packed = is_packed(&ast.attrs);
46        let mut copy_types: Vec<&Type> = Vec::new();
47
48        let mut hash_token_stream = proc_macro2::TokenStream::new();
49
50        if let Data::Struct(data) = &ast.data {
51            let this = quote_mixed!(self);
52
53            let built_in_hash: ExprPath =
54                syn::parse2(quote_mixed!(::core::hash::Hash::hash)).unwrap();
55
56            for (index, field) in data.fields.iter().enumerate() {
57                let field_attribute = FieldAttributeBuilder {
58                    enable_ignore: true,
59                    enable_method: true,
60                }
61                .build_from_attributes(&field.attrs, traits)?;
62
63                if field_attribute.ignore {
64                    continue;
65                }
66
67                let field_name = if let Some(ident) = field.ident.as_ref() {
68                    IdentOrIndex::from(ident)
69                } else {
70                    IdentOrIndex::from(index)
71                };
72
73                if is_packed {
74                    copy_types.push(&field.ty);
75                }
76
77                let hash = field_attribute.method.as_ref().unwrap_or_else(|| {
78                    hash_types.push(&field.ty);
79                    &built_in_hash
80                });
81
82                let field_ref = borrow_field(is_packed, &this, &field_name);
83
84                hash_token_stream.extend(quote_mixed!( #hash(#field_ref, #state); ));
85            }
86        }
87
88        let ident = &ast.ident;
89        let hasher_ident = bindings.fresh("H");
90
91        let packed_copy_predicates = if is_packed {
92            type_attribute.bound.packed_copy_predicates(&ast.generics, &copy_types, &ast.ident)
93        } else {
94            WherePredicates::new()
95        };
96
97        let mut bound =
98            type_attribute.bound.into_where_predicates_by_generic_parameters_check_types(
99                &ast.generics,
100                &syn::parse2(quote_mixed!(::core::hash::Hash)).unwrap(),
101                &hash_types,
102                &ast.ident,
103                &BOUND_EXCEPTIONS_HASH,
104            );
105
106        extend_where_predicates(&mut bound, packed_copy_predicates);
107
108        let generics = crate::common::generics::with_predicates(ast.generics.clone(), bound);
109
110        let (impl_generics, ty_generics, where_clause) = generics.split_for_impl();
111
112        token_stream.extend(bindings.wrap(quote_mixed! {
113            #generated_impl_attributes
114            impl #impl_generics ::core::hash::Hash for #ident #ty_generics #where_clause {
115                #[inline]
116                fn hash<#hasher_ident: ::core::hash::Hasher>(&self, #state: &mut #hasher_ident) {
117                    #hash_token_stream
118                }
119            }
120        }));
121
122        Ok(())
123    }
124}
