6/// Builds the attributes that every generated impl carries.
7pub(crate) fn generated_impl_attributes(attributes: &[Attribute]) -> proc_macro2::TokenStream {
8    let lints = generated_lint_attributes(attributes);
9    quote!(#[automatically_derived] #lints)
10}

12/// Copies `allow`, `warn`, and `deny` from the input type.
13/// An `expect` becomes `allow` because a generated item may not trigger the original lint.
14pub(crate) fn generated_lint_attributes(attributes: &[Attribute]) -> proc_macro2::TokenStream {
15    let mut token_stream = proc_macro2::TokenStream::new();
16
17    for attribute in attributes {
18        let path = attribute.path();
19
20        if path.is_ident("expect") {
21            // A malformed `expect` is reported by the compiler on the user's own item, so there is nothing to copy here.
22            if let Meta::List(list) = &attribute.meta {
23                let lints = &list.tokens;
24
25                // The `allow` ident is created here instead of being taken from the user's attribute, so that lints looking for hand-written `allow` attributes, such as `clippy::allow_attributes`, still see this one as macro generated and skip it.
26                token_stream.extend(quote!(#[allow(#lints)]));
27            }
28        } else if path.is_ident("allow") || path.is_ident("warn") || path.is_ident("deny") {
29            attribute.to_tokens(&mut token_stream);
30        }
31    }
32
33    token_stream
34}

36/// Returns true if the input type is `#[repr(packed)]` or `#[repr(packed(N))]`.
37///
38/// Every packing level is treated the same way, matching the built-in derives: a field's alignment cannot be worked out from the type syntax, so the generated code has to assume that it exceeds the packing.
39pub(crate) fn is_packed(attributes: &[Attribute]) -> bool {
40    for attribute in attributes {
41        if attribute.path().is_ident("repr")
42            && let Meta::List(list) = &attribute.meta
43            && let Ok(items) = list.parse_args_with(Punctuated::<Meta, Token![,]>::parse_terminated)
44        {
45            // A malformed `repr` is reported by the compiler on the user's own item, so a parse failure is simply skipped here.
46            for item in items {
47                if item.path().is_ident("packed") {
48                    return true;
49                }
50            }
51        }
52    }
53
54    false
55}

57/// Builds an expression that borrows a field for reading.
58///
59/// A field of a `#[repr(packed)]` type cannot be borrowed in place, so its value is copied into a temporary first, which requires the field type to implement `Copy` exactly like the built-in derives do.
60pub(crate) fn borrow_field<T: ToTokens>(
61    is_packed: bool,
62    receiver: &proc_macro2::TokenStream,
63    field_name: &T,
64) -> proc_macro2::TokenStream {
65    if is_packed {
66        quote_mixed!(&{ #receiver.#field_name })
67    } else {
68        quote_mixed!(&#receiver.#field_name)
69    }
70}
