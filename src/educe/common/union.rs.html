5/// Builds the statement that binds the whole storage of a union as a byte slice.
6///
7/// A union does not track its active field at runtime, so the unsafe union derives read its storage as bytes instead of reading a field.
8pub(crate) fn union_bytes(binding: &Ident, receiver: &TokenStream) -> TokenStream {
9    quote_mixed!(
10        // SAFETY: The live union reference provides a valid pointer and size; the unsafe derive contract requires every byte, including padding and bytes outside the active field, to be initialized and unchanged during this call.
11        // The user must preserve this condition after every construction, write, move, and copy; reading uninitialized bytes is undefined behavior.
12        let #binding = unsafe {
13            ::core::slice::from_raw_parts(
14                #receiver as *const Self as *const ::core::primitive::u8,
15                ::core::mem::size_of::<Self>(),
16            )
17        };
18    )
19}
