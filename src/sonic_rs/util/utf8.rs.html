3// simduft8 will cause `out-of-bounds pointer arithmetic` when using Miri tests
4#[cfg(not(miri))]
5#[inline]
6pub(crate) fn from_utf8(data: &[u8]) -> Result<&str> {
7    simdutf8::basic::from_utf8(data).or_else(|_| from_utf8_compat(data))
8}
9
10#[cfg(miri)]
11pub(crate) fn from_utf8(data: &[u8]) -> Result<&str> {
12    std::str::from_utf8(data)
13        .map_err(|e| Error::syntax(ErrorCode::InvalidUTF8, data, e.valid_up_to()))
14}
15
16#[cfg(not(miri))]
17#[cold]
18fn from_utf8_compat(data: &[u8]) -> Result<&str> {
19    // compat::from_utf8 is slower than basic::from_utf8
20    simdutf8::compat::from_utf8(data)
21        .map_err(|e| Error::syntax(ErrorCode::InvalidUTF8, data, e.valid_up_to()))
22}
