1use std::path::Path;
2use std::sync::Arc;
3
4use crate::{ClientState, Error};
5
6/// A reduced, but public, version of ReadDirSpec
7pub struct ReadChildren<C: ClientState> {
8    /// Path that will be used to read child entries. This is
9    /// automatically set for directories.
10    pub(crate) path: Arc<Path>,
11    /// If the resulting `fs::read_dir` generates an error
12    /// then that error is stored here.
13    pub(crate) error: Option<Error>,
14    /// Use this to customize the ReadDirState passed to the next
15    /// process_read_dir.
16    /// If None, will clone the previous ReadDirState after the parent
17    /// call to process_read_dir.
18    pub client_read_state: Option<C::ReadDirState>,
19}
20
21impl<C: ClientState> ReadChildren<C> {
22    pub(crate) fn new(path: &Path) -> Self {
23        Self {
24            path: Arc::from(path),
25            error: None,
26            client_read_state: None,
27        }
28    }
29
30    /// Return the error stored that occurred when reading the directory.
31    pub fn error(&self) -> Option<&Error> {
32        self.error.as_ref()
33    }
34}
