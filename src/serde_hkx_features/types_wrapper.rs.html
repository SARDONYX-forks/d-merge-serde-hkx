3use crate::ClassMap;
4use havok_classes::Classes;
5use havok_types::Pointer;
6use indexmap::IndexMap;
7use rayon::prelude::*;
8use serde::{Deserialize, Serialize};
9use std::borrow::Cow;
10
11/// - key: class index(e.g `#0001`)
12/// - value: C++ Class
13#[cfg_attr(feature = "json_schema", derive(schemars::JsonSchema))]
14#[derive(Debug, Clone, Default, PartialEq, Serialize, Deserialize)]
15pub struct ClassPtrMap<'a> {
16    /// Path of Json schema
17    #[cfg_attr(feature = "json_schema", schemars(rename = "$schema"))]
18    #[serde(default = "default_schema_url")]
19    schema: Cow<'a, str>,
20    #[serde(flatten)]
21    #[serde(borrow)]
22    classes: IndexMap<Pointer<'a>, Classes<'a>>,
23}
24
25#[inline]
26const fn default_schema_url() -> Cow<'static, str> {
27    Cow::Borrowed(
28        "https://raw.githubusercontent.com/SARDONYX-sard/serde-hkx/refs/tags/0.7.0/assets/serde_hkx_schema.json",
29    )
30}
31
32impl<'a> ClassPtrMap<'a> {
33    pub fn from_class_map(value: ClassMap<'a>) -> Self {
34        Self {
35            schema: default_schema_url(),
36            classes: value
37                .into_par_iter()
38                .map(|(k, v)| (Pointer::new(k), v))
39                .collect(),
40        }
41    }
42
43    pub fn into_class_map(self) -> ClassMap<'a> {
44        self.classes
45            .into_par_iter()
46            .map(|(k, v)| (k.into_inner(), v))
47            .collect()
48    }
49}
