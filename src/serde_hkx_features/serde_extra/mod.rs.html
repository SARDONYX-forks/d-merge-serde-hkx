10    /// Serialize bytes(file contents) to a file.
11    ///
12    /// # Errors
13    /// If the information required for serialization is missing.
14    ///
15    /// See `serde_hkx::errors::ser::Error` for possible errors that may occur.
16    ///
17    /// # Panics
18    /// If the `OutFormat` is not an enabled extra format (`json`).
19    /// That means the API is being used incorrectly.
20    pub fn to_bytes(classes: &mut ClassPtrMap<'_>, output_format: Format) -> Result<Vec<u8>> {
21        let contents = match output_format {
22            #[cfg(feature = "json")]
23            Format::Json => sonic_rs::to_string_pretty(&classes).context(JsonSerSnafu)?,
24            _ => unreachable!(),
25        };
26        Ok(contents.into_bytes())
27    }
