1//! `serde_hkx_features`'s errors
2//!
3//! # Path policy
4//! Errors do not carry the path of the file being processed.
5//! The caller that knows the path (e.g. CLI) attaches it with [`Error::with_path`].
6//!
7//! Only operations that handle multiple paths internally (dir conversion/verify, diff)
8//! attach the path themselves.
9use snafu::Location;
10use std::path::{Path, PathBuf};

12#[cfg(not(feature = "json"))]

14#[cfg(feature = "json")]
15const SUPPORTED_EXTENSIONS: &str = "`.hkx`, `.xml`, `.json`";

172/// Run `f` and convert a panic into [`Error::Panicked`].
173///
174/// # Why need this?
175/// In parallel processing of dir, a panic of one file stops the progress count of the file,
176/// so `success + failure` no longer matches the total.
177///
178/// # Errors
179/// Error of `f` or [`Error::Panicked`]
180pub fn catch_panic<T, F>(f: F) -> Result<T>
181where
182    F: FnOnce() -> Result<T>,
183{
184    std::panic::catch_unwind(std::panic::AssertUnwindSafe(f)).unwrap_or_else(|payload| {
185        let msg = payload
186            .downcast_ref::<&str>()
187            .map(|s| (*s).to_string())
188            .or_else(|| payload.downcast_ref::<String>().cloned())
189            .unwrap_or_else(|| "unknown panic payload".to_string());
190        Err(Error::Panicked { msg })
191    })
192}

194impl Error {
195    /// Wrap this error with the path of the file being processed.
196    #[inline]
197    pub fn with_path<P>(self, path: P) -> Self
198    where
199        P: AsRef<Path>,
200    {
201        Self::WithPath {
202            path: path.as_ref().to_path_buf(),
203            source: Box::new(self),
204        }
205    }
206}

208fn display_errors(errors: &[(PathBuf, Error)]) -> String {
209    use core::fmt::Write as _;
210
211    let mut out = String::new();
212    for (path, err) in errors {
213        let _ = writeln!(out, "- {}:\n  {err}", path.display());
214    }
215    out
216}

218/// `Result` for `serde_hkx_features` crate.
