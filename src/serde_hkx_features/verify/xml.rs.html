1//! hkx -> XML -> hkx round trip verification.
2//!
3//! hkx -> XML is lossy by the specification of Havok XML.
4//! - float: `{:.06}`
5//! - `Matrix3`/`Rotation`/`QsTransform`/`Transform`: `w` of each `Vector4` is not written.
6//!
7//! So comparing bytes(`hkx -> XML -> hkx` == `hkx`) can't detect XML bugs. Instead, check the followings.
8//!
9//! 1. `xml_fixed_point`: `XML(A)` == `XML(from_xml(XML(A)))`
10//!    - Detects the information that is written to XML but not read(or read differently).
11//! 2. `xml_semantic`: `A`(from hkx) == `B`(from XML) except for the above lossy parts.
12//!    - Detects the information that is not written to XML(or written wrongly).
13//! 3. `xml_to_bytes`: `B` == `from_hkx(to_hkx(B))` strictly.
14//!    - Detects the bytes (de)serializer bugs for the classes from XML.(e.g. pointer order, sorting)
15use super::debug_cmp::{CmpOptions, compare};
16use crate::{
17    ClassMap, diff,
18    error::{HkxDeSnafu, HkxSerSnafu, Result, XmlDeSnafu, XmlRoundTripMismatchSnafu, XmlSerSnafu},
19};
20use serde_hkx::{HavokSort as _, bytes::serde::hkx_header::HkxHeader};
21use snafu::ResultExt as _;
22use std::{collections::HashMap, path::Path};

24/// Verify hkx -> XML -> hkx round trip of a hkx file.
25///
26/// # Errors
27/// - Failed to (de)serialize.
28/// - [`crate::error::Error::XmlRoundTripMismatch`]
29pub fn verify_xml_file<P>(path: P, color: bool) -> Result<()>
30where
31    P: AsRef<Path>,
32{
33    let bytes = std::fs::read(path)?;
34    verify_xml_bytes(&bytes, color)
35}

37/// Verify hkx -> XML -> hkx round trip of hkx bytes.
38///
39/// # Errors
40/// - Failed to (de)serialize.
41/// - [`crate::error::Error::XmlRoundTripMismatch`]
42pub fn verify_xml_bytes(bytes: &[u8], color: bool) -> Result<()> {
43    let header = HkxHeader::from_bytes(bytes).context(HkxDeSnafu)?;
44    let to_xml = |classes: &mut ClassMap| -> Result<String> {
45        let top_ptr = classes.sort_for_xml().context(XmlSerSnafu)?;
46        serde_hkx::to_string(classes, &top_ptr, &header).context(XmlSerSnafu)
47    };
48
49    // A: hkx -> classes
50    let mut a: ClassMap = serde_hkx::from_bytes(bytes).context(HkxDeSnafu)?;
51    let xml_a = to_xml(&mut a)?;
52
53    // B: XML -> classes
54    let mut b: ClassMap = serde_hkx::from_str(&xml_a).context(XmlDeSnafu)?;
55    let xml_b = to_xml(&mut b)?;
56
57    // 1. XML fixed point
58    if xml_a != xml_b {
59        return XmlRoundTripMismatchSnafu {
60            check: "xml_fixed_point",
61            detail: diff::diff(&xml_a, &xml_b, color),
62        }
63        .fail();
64    }
65
66    // 2. hkx classes == XML classes (except for lossy parts)
67    let opt = CmpOptions {
68        float_tolerance: true,
69        ignore_xml_lost_w: true,
70    };
71    if let Err(detail) = compare_maps(&a, &b, opt, None) {
72        return XmlRoundTripMismatchSnafu {
73            check: "xml_semantic",
74            detail,
75        }
76        .fail();
77    }
78
79    // 3. XML classes -> hkx -> classes(strict)
80    let len = b.len();
81    b.sort_for_bytes();
82    if b.len() != len {
83        let dropped: Vec<_> = a.keys().filter(|key| !b.contains_key(*key)).collect();
84        return XmlRoundTripMismatchSnafu {
85            check: "xml_to_bytes",
86            detail: format!("`sort_for_bytes` dropped unreferenced classes: {dropped:?}"),
87        }
88        .fail();
89    }
90    let bytes_b = serde_hkx::to_bytes(&b, &header).context(HkxSerSnafu)?;
91    let c: ClassMap = serde_hkx::from_bytes(&bytes_b).context(HkxDeSnafu)?;
92    if b.len() != c.len() {
93        return XmlRoundTripMismatchSnafu {
94            check: "xml_to_bytes",
95            detail: format!("class count: {} -> {}", b.len(), c.len()),
96        }
97        .fail();
98    }
99    // Pointer names are renumbered in the order of bytes.
100    let rename: HashMap<String, String> = b
101        .keys()
102        .zip(c.keys())
103        .map(|(b, c)| (b.to_string(), c.to_string()))
104        .collect();
105    let pairs = b
106        .iter()
107        .zip(c.iter())
108        .map(|((b_key, b), (c_key, c))| (Entry::new(b_key, b), Entry::new(c_key, c)));
109    if let Err(detail) = compare_pairs(pairs, CmpOptions::default(), Some(&rename)) {
110        return XmlRoundTripMismatchSnafu {
111            check: "xml_to_bytes",
112            detail,
113        }
114        .fail();
115    }
116
117    Ok(())
118}

120/// Compare classes that have the same keys.
121fn compare_maps(
122    left: &ClassMap,
123    right: &ClassMap,
124    opt: CmpOptions,
125    rename: Option<&HashMap<String, String>>,
126) -> Result<(), String> {
127    if left.len() != right.len() || left.keys().any(|k| !right.contains_key(k)) {
128        let missing: Vec<_> = left.keys().filter(|k| !right.contains_key(*k)).collect();
129        let extra: Vec<_> = right.keys().filter(|k| !left.contains_key(*k)).collect();
130        return Err(format!(
131            "class keys differ: missing {missing:?}, extra {extra:?}"
132        ));
133    }
134    let pairs = left.iter().filter_map(|(key, l)| {
135        let (r_key, r) = right.get_key_value(key)?;
136        Some((Entry::new(key, l), Entry::new(r_key, r)))
137    });
138    compare_pairs(pairs, opt, rename)
139}

141/// (key, class name, `Debug` string)
142struct Entry {
143    key: String,
144    name: &'static str,
145    debug: String,
146}
147
148impl Entry {
149    fn new(key: &str, class: &havok_classes::Classes<'_>) -> Self {
150        use havok_serde::HavokClass as _;
151        Self {
152            key: key.to_string(),
153            name: class.name(),
154            debug: format!("{class:?}"),
155        }
156    }
157}
158
159fn compare_pairs(
160    pairs: impl Iterator<Item = (Entry, Entry)>,
161    opt: CmpOptions,
162    rename: Option<&HashMap<String, String>>,
163) -> Result<(), String> {
164    for (l, r) in pairs {
165        if let Err(mismatch) = compare(&l.debug, &r.debug, opt, rename) {
166            return Err(format!("{} -> {}({}): {mismatch}", l.key, r.key, l.name));
167        }
168    }
169    Ok(())
170}
